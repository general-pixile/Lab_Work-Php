<?php

// Sample string
$text = "Hello World, Welcome to PHP Programming";

// 1) strlen() - Find the length of a string
echo "1. Length of string: " . strlen($text) . "<br>";

// 2) strpos() - Find the position of a specific word
echo "2. Position of 'World': " . strpos($text, "World") . "<br>";

// 3) str_word_count() - Count the number of words
echo "3. Number of words: " . str_word_count($text) . "<br>";

// 4) strrev() - Reverse the string
echo "4. Reversed string: " . strrev($text) . "<br>";

// 5) strtolower() - Convert string to lowercase
echo "5. Lowercase string: " . strtolower($text) . "<br>";

// 6) strtoupper() - Convert string to uppercase
echo "6. Uppercase string: " . strtoupper($text) . "<br>";

?>
