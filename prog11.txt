 Write a PHP Program to reverse an array values entered by user.


<!DOCTYPE html>
<html>
<body>

<form method="post">
    Enter array values separated by commas:<br><br>
    <input type="text" name="arr" required>
    <input type="submit" value="Reverse Array">
</form>

<?php
if(isset($_POST['arr']))
{
    $arr = explode(",", $_POST['arr']);

    echo "<h3>Original Array:</h3>";
    print_r($arr);

    $reversed = array_reverse($arr);

    echo "<h3>Reversed Array:</h3>";
    print_r($reversed);
}
?>

</body>
</html>
